cmake_minimum_required(VERSION 2.8.3)
project(rgbd_following)
find_package(catkin REQUIRED COMPONENTS rospy roscpp costmap_2d pluginlib sensor_msgs std_msgs cv_bridge message_filters geometry_msgs nav_msgs tf2_ros actionlib move_base_msgs)
catkin_python_setup()
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_library(rgbd_commissioning_clearance src/commissioning_clearance_layer.cpp)
set_target_properties(rgbd_commissioning_clearance PROPERTIES COMPILE_FLAGS "-std=c++11")
target_link_libraries(rgbd_commissioning_clearance ${catkin_LIBRARIES})
install(TARGETS rgbd_commissioning_clearance LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES clearance_plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
catkin_install_python(PROGRAMS scripts/rgbd_probe.py scripts/synthetic_rgbd.py scripts/rgbd_target.py scripts/rgbd_follower.py scripts/rgbd_velocity_gate.py scripts/rgbd_trial_recorder.py scripts/rgbd_identity_preview.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/navigation_smoke.test)
  add_rostest(test/clearing_observations.test)
  add_rostest(test/commissioning_clearance.test)
endif()
